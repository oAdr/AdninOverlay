cmake_minimum_required(VERSION 3.21)
project(Adnin LANGUAGES CXX)
enable_language(RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use an x64 Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(AdninInjector WIN32 src/injector/main.cpp src/injector/result-ui.cpp src/injector/payload.cpp src/injector/runtime-profile.cpp
  "${CMAKE_BINARY_DIR}/generated/adnin-payload.rc")
set_target_properties(AdninInjector PROPERTIES OUTPUT_NAME Adnin)
target_include_directories(AdninInjector PRIVATE "${CMAKE_BINARY_DIR}/generated" src/injector)
target_compile_definitions(AdninInjector PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(AdninInjector PRIVATE user32 gdi32 shell32 ole32 uuid dwmapi bcrypt)
if(MSVC)
  target_compile_options(AdninInjector PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(AdninInjector PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(AdninInjector PRIVATE -municode -static -Wl,--dynamicbase,--nxcompat,--high-entropy-va,--no-insert-timestamp)
endif()
enable_testing()
add_test(NAME injector-help COMMAND "$<TARGET_FILE:AdninInjector>" --help)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(ADNIN_TEST_WORK_DIRECTORY "${CMAKE_BINARY_DIR}/test-work/injector" CACHE PATH "Private directory for offline injector fixtures")
add_executable(injector-ui-host tests/injector-ui-host.cpp src/injector/result-ui.cpp src/injector/payload.cpp src/injector/runtime-profile.cpp
  "${CMAKE_BINARY_DIR}/generated/adnin-payload.rc")
target_include_directories(injector-ui-host PRIVATE "${CMAKE_BINARY_DIR}/generated" src/injector)
target_compile_definitions(injector-ui-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(injector-ui-host PRIVATE user32 gdi32 shell32 ole32 uuid dwmapi bcrypt)
if(MSVC)
  target_compile_options(injector-ui-host PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(injector-ui-host PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(injector-ui-host PRIVATE -municode -static)
endif()
add_test(NAME injector-offline-ui-and-cache COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_injector.py" --exe "$<TARGET_FILE:AdninInjector>"
  --host "$<TARGET_FILE:injector-ui-host>" --dll "${CMAKE_BINARY_DIR}/bin/Adnin.dll"
  --compat-dll "${CMAKE_BINARY_DIR}/bin/AdninVanilla.dll"
  --work "${ADNIN_TEST_WORK_DIRECTORY}" -v)
add_executable(runtime-profile-test tests/runtime-profile.cpp src/injector/runtime-profile.cpp)
target_include_directories(runtime-profile-test PRIVATE src/injector)
target_compile_definitions(runtime-profile-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(runtime-profile-test PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(runtime-profile-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(runtime-profile-test PRIVATE -static)
endif()
add_test(NAME runtime-profile-selection COMMAND "$<TARGET_FILE:runtime-profile-test>")
add_executable(thread-selection-test tests/thread-selection.cpp)
add_executable(dummy-host tests/dummy-host.cpp)
if(NOT MSVC)
  target_compile_options(thread-selection-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(thread-selection-test PRIVATE -municode -static)
  target_link_options(dummy-host PRIVATE -municode -static)
endif()
add_test(NAME thread-selection-regression COMMAND "$<TARGET_FILE:thread-selection-test>" "${CMAKE_BINARY_DIR}/bin/Adnin.dll")
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
add_test(NAME injection-rejects-uninitialized-runtime COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/controlled-load.ps1"
  -Injector "$<TARGET_FILE:AdninInjector>" -Dll "${CMAKE_BINARY_DIR}/bin/Adnin.dll"
  -DummyHost "$<TARGET_FILE:dummy-host>" -Report "${CMAKE_BINARY_DIR}/controlled-load.json")
